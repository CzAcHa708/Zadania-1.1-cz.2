    char tab[8][8];

    for (int i = 0; i < 8; i++) {
        for (int j = 0; j < 8; j++) {
            if ((i + j) % 2 == 0)
                tab[i][j] = '#';
            else
                tab[i][j] = '.';

            cout << tab[i][j] << " ";
        }
        cout << endl;
    }

    return 0;
}

Ćwiczenie 3.3
a)  struct A { int x; int y; };                 = __8 bajtów (4 + 4)____

b)  struct B { char c; int x; };                = __8 bajtów (1 + 3 bajty wyrównania + 4)____

c)  struct C { char c; char d; int x; };        = __8 bajtów (1 + 1 + 2 bajty wyrównania + 4)____

d)  union  D { int x; double y; char c[4]; };   = __8 bajtów (największe pole ma 8 bajtów)____

int wymaga wyrównania do 4 bajtów. Po char dodawane są 3 bajty wypełnienia, aby int zaczynał się pod odpowiednim adresem.

struct Pomiar {
    int typ; // 1 - całkowita, 2 - rzeczywista, 3 - błąd

    union {
        int calkowita;
        double rzeczywista;
        char blad[50];
    } wynik;
};

if (pomiar.typ == 1)
    cout << pomiar.wynik.calkowita;
else if (pomiar.typ == 2)
    cout << pomiar.wynik.rzeczywista;
else if (pomiar.typ == 3)
    cout << pomiar.wynik.blad;
